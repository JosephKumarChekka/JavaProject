InsertApp.java
--------------
package com.rt.www;

import org.bson.Document;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;

public class InsertApp 
{
	public static void main(String[] args) 
	{
		try(MongoClient mongoClient = MongoClients.create("mongodb://localhost:27017");)
		{
			MongoDatabase mongoDatabase = mongoClient.getDatabase("myDB");
			MongoCollection<Document> mongoCollection = mongoDatabase.getCollection("products");
			Document doc = new Document("pid",101).append("pname", "Lenovo").append("pprice", 60000);
			mongoCollection.insertOne(doc);
			System.out.println("Record Inserted");
		}
		catch(Exception e)
		{
			e.printStackTrace();
		}
	}
}

SelectApp.java

package com.rt.www;

import org.bson.Document;

import com.mongodb.client.MongoClient;
import com.mongodb.client.MongoClients;
import com.mongodb.client.MongoCollection;
import com.mongodb.client.MongoDatabase;

public class SelectApp 
{
	public static void main(String[] args) 
	{
		try(MongoClient mongoClient = MongoClients.create("mongodb://localhost:27017");)
		{
			MongoDatabase mongoDatabase = mongoClient.getDatabase("myDB");
			
			MongoCollection<Document> mongoCollection = mongoDatabase.getCollection("products");
			
			Document document = mongoCollection.find(new Document("pid",101)).first();
			
			System.out.println(document);
			
		}
	}
}
