Q) Write a java program to perform swapping of two numbers?

Input:
	a=10 
	b=20

Output:
	Before swapping a=10 and b=20 
	After swapping a=20 and b=10 

import java.util.Scanner;
class Example7
{
	public static void main(String[] args)
	{
		Scanner sc = new Scanner(System.in);
		
		System.out.println("Enter two numbers :");
		int a = sc.nextInt(); // 10
		int b = sc.nextInt(); // 20 
		
		System.out.println("Before swapping a="+a+" and b="+b);
		
		//logic 
		int temp = a;
		a = b;
		b = temp;
		
		System.out.println("After swapping a="+a+" and b="+b);
	}
}
				
Q) Write a java program to perform swapping of two numbers without using third variable?

Input:
	a=10 
	b=20

Output:
	Before swapping a=10 and b=20 
	After swapping a=20 and b=10 

import java.util.Scanner;
class Example8
{
	public static void main(String[] args)
	{
		Scanner sc = new Scanner(System.in);
		
		System.out.println("Enter two numbers :");
		int a = sc.nextInt(); // 10
		int b = sc.nextInt(); // 20 
		
		System.out.println("Before swapping a="+a+" and b="+b);
		
		//logic 
		a = a + b;
		b = a - b;
		a = a - b;
		
		System.out.println("After swapping a="+a+" and b="+b);
	}
}
				

Q) Write a java program to accept person's date of birth and find out his age?

 
import java.time.LocalDate;
import java.time.Period;
class Example9
{
	public static void main(String[] args)
	{
		LocalDate dob = LocalDate.of(2007,10,18);
		System.out.println(dob);
		
		LocalDate date = LocalDate.now();
		
		Period p = Period.between(dob,date);
		
		int age = p.getYears();
		
		System.out.println("You are "+age+" years old");
	}
}
				

Q) Write a java program to convert time to seconds?

Input:
	Hours : 10
	Minutes : 30
	Seconds	: 23
Output:
	37823


class Example10
{
	public static void main(String[] args)
	{
		int hours = 10;
		int minutes = 30;
		int seconds = 23;
		
		int totalSeconds = (hours*3600) + (minutes*60) + seconds;
		
		System.out.println(totalSeconds);
	}
}
